def tables():
    a = json.loads((ROOT / 'ANALYSIS.json').read_text())
    p = json.loads((ROOT / 'protocol.json').read_text())
    by = {(r['case'], r['answer_model']): r for r in a['conditions']}
    models = p['answer_models']
    result = ['| Case | arc:lite | arc:chat | arc:apex |', '| --- | --- | --- | --- |']
    grades = result.copy()
    for c in p['cases']:
        cid = c['case_id']; label = 'N04' if cid == 'N04_INITIAL' else cid
        cells, scores = [], []
        for model in models:
            r = by[cid, model]
            missing = [LABELS[g['parent_fact_id']] for g in c['coverage_groups'] if g['source_text'] in r['missing_groups']]
            cells.append(str(r['groups_obtained']) + '/' + str(r['groups_available']) + '; ' + (', '.join(missing) if missing else 'complete'))
            scores.append(' / '.join(str(r[k]) if r[k] is not None else 'N/A' for k in ['D5c', 'D3a']))
        result.append('| [' + label + ' — ' + NAMES[cid] + '](case_outputs/' + cid + '.txt) | ' + ' | '.join(cells) + ' |')
        grades.append('| ' + label + ('*' if cid in ['C05', 'N02'] else '') + ' | ' + ' | '.join(scores) + ' |')
    print('INFORMATION OBTAINED AND MISSED\n' + '\n'.join(result))
    print('\nSCORES: D5c / D3a\n' + '\n'.join(grades))
    for model in models:
        rs = [r for r in a['conditions'] if r['answer_model'] == model]
        print(model, 'information', sum(r['groups_obtained'] for r in rs), '/54',
              'turns', sum(r['turns'] for r in rs),
              'word-limit exceptions', [(r['case'], r['word_count']) for r in rs if not r['word_count']['within_range']],
              'resources', a['resources'][model])


if __name__ == '__main__':
    tables()
