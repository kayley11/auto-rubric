def read(p):return json.loads(p.read_text())
old={c['id']:c for c in read(PILOT/'data/cases.json')}
refs=read(PILOT/'data/reference_notes.json')
new={}
for pack in ['new_cases_20261022','new_cases_batch2_20261022']:
 rr={c['case_id']:c['reference'] for c in read(PILOT/pack/'reference_notes.json')['cases']}
 for c in read(PILOT/pack/'cases.json')['cases']:
  new[c['case_id']]={**c,'reference':rr[c['case_id']],'pack':pack}

TOPICS={
 'presenting_symptoms':'Current symptoms and associated symptoms',
 'timecourse':'Onset, course, triggers and aggravating factors',
 'previous_episodes':'Earlier similar or other relevant episodes',

 'tests':'Existing examination findings and test reports; no new test is performed',
 'exposures':'Travel, exposures and relevant lifestyle history',

CASES=[]

def add(cid,opening,groups,reason,notes=''):
 c=old[cid] if cid in old else new[cid]
 facts=c['facts'];atoms=[]
 for topic,channel,key,quote in groups:
  assert quote in facts,(cid,quote)
  atoms.append({'id':f'{cid}_F{len(atoms)+1:02d}','topic':topic,'channel':channel,
   'coverage_target':key,'text':quote})
 ref=json.dumps(refs[cid],ensure_ascii=False,indent=2) if cid in old else c['reference']
 task=c['task']
 if cid=='C05':
  task=task.replace('what the normal scan does and does not establish','what any investigation findings obtained do and do not establish')
 if cid=='N02':
  task=task.replace('what does the EEG result tell us','what do any investigation results obtained tell us')
  task=task.replace('Explain the relevance of the low glucose without claiming a final cause','Explain the relevance of any glucose measurements obtained without claiming a final cause')
 if cid=='N04_INITIAL':
  task=task.replace('Consider the recurrence, CSF pattern, posture-related headache, prior treatment and imaging together.', 'Consider the history, treatment and investigation information actually supplied together.')
 if cid=='N05':
  task=task.replace('Explain how the gastrointestinal, neurological and cardiac findings affect the working differential.', 'Explain how the symptom and investigation findings actually obtained affect the working differential.')
  task=task.replace('Interpret what the PYP result supports and what it does not establish, alongside the other investigations.', 'Interpret what the investigation results actually obtained support and what they do not establish.')
  task=task.replace("Address the significance of the earlier abnormal vital signs without assuming that they describe the patient's current condition.", 'Distinguish historical observations actually obtained from the current state; do not assume current vital signs.')
 selected=['D1','D2a','D2d','D5u','D7c']
 if cid not in ['C05','N02']:selected.append('D3a')
 else:selected.append('D7b')
 if cid in ['C01','C02','C03','C04','C05','N02']:selected.append('D6b')
 CASES.append({'case_id':cid,'title':c.get('topic',c.get('title')),
  'source_case_id':cid,'eligibility_reason':reason,'adaptation_note':notes,
  'full_visible_snapshot':facts,'opening':opening,'facts':atoms,'final_task':task,'original_final_task':c['task'],
  'reference':ref,'reference_status':'draft — awaiting clinical confirmation',
  'final_criteria':selected,'dialogue_criteria':['D5c','D5u','D7b'],
  'historical_run':('runs/apex_v1_5/results/operations/'+cid+'.generate.json') if cid in old else
   c['pack']+'/runs/'+('ui_batch_20261022' if c['pack']=='new_cases_20261022' else 'ui_batch2_20261022')+'/'+cid+'.json'})

protocol={'prepared_at':datetime.now(timezone.utc).isoformat(),'version':'patient-agent-v2-bounded-speech',
 'instruction_adaptation':'Remove result-specific hints from final task wording where needed, identically in both matched arms. Original task wording is retained in each case. The interview agent is not shown the final clinical task before asking questions.',
 'purpose':'Compare full-information static answers with question-driven information acquisition, not a test proving removal of training contamination.',
 'answer_model':'arc:lite','patient_model':'arc:lite','judge_model':'arc:apex','reference_status':'draft',
 'maximum_question_turns':6,'minimum_question_turns':0,'topics':TOPICS,
 'routing':'The doctor returns a natural-language question plus at most two standard, case-independent topic tags. The controller releases only matching fixed facts. Tags and actual questions are saved for inspection; this is assisted routing, not a validated unrestricted natural-language matcher.',
 'surface_policy':'Fixed, source-checked first-person sentences. The patient LLM may order only the permitted IDs; the controller renders the frozen sentences and no other clinical text. If layout JSON fails twice, use the source order and record the fallback. Free paraphrasing was rejected after the separately archived C02 preflight added unsupported negatives and changed chronology.',
 'patient_boundary':'The patient LLM receives only the opening, current question, and permitted patient-reportable facts; it never receives unasked facts, source title, rubric, reference, diagnosis or outcomes. Existing report/collateral excerpts are delivered verbatim by the controller, separately from patient speech.',
 'unknown_policy':'Unspecified facts remain unavailable. Never turn an absent source field into a negative or normal finding.',
 'coverage_policy':'Fact IDs mark released information. Coverage targets are researcher-prepared information groups, not clinician-approved mandatory points. Count a target only when its complete quoted content has been supplied; an opening may reveal part of a group. Report unelicited groups as not fully obtained, not wholly unknown. Released does not mean correctly used in the final answer.',
 'premature_conclusion_policy':'Record the first tentative hypothesis and any explicit established/confirmed diagnostic claim. A claim requiring unprovided evidence is flagged for clinical review, not automatically declared misdiagnosis.',
 'comparison':'New matched static and interactive final answers use the same snapshot, final task, word-count rule, models and selected original rubric anchors. Static sees all fixed facts at the start; interactive sees only elicited facts. No total score or causal claim of reduced memorisation.',
 'word_count':'Retain headings, remove Markdown markers, whitespace count, hyphenated terms count once; no added tolerance. Give the deterministic count to D7c.',
 'scoring':'Shared final-answer criteria plus task-specific summary/audience/safety items. Separately grade the complete dialogue for context seeking, uncertainty and patient-facing communication. Unselected items remain unassessed.',
 'repeats':'One successful visible output per operation; save failed technical attempts, at most two attempts per operation. Do not regenerate clinically weak or overlength answers.',
