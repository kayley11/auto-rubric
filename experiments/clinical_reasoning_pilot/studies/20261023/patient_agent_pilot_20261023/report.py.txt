def read(path):
    return json.loads(path.read_text())


def dump(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def cell(value):
    return str(value).replace("|", "\\|").replace("\n", " ")


def table(headers, rows):
    return ["| " + " | ".join(map(cell, headers)) + " |",
            "| " + " | ".join("---" for _ in headers) + " |"] + [
                "| " + " | ".join(map(cell, row)) + " |" for row in rows]


def write_csv(path, rows):
    if not rows:
        return
    with path.open("w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0]))
        w.writeheader()
        w.writerows(rows)


def label(grade):
    return str(grade["score"]) if grade.get("score") is not None else grade["status"]


def title(c):
    return re.sub(r"^N\d+\s+—\s+", "", c["title"])


def totals(calls):
    attempts = [a for c in calls for a in c["attempts"]]
    usages = [a.get("response", {}).get("usage") for a in attempts]
    known = [u for u in usages if isinstance(u, dict)]
    return {
        "operations": len(calls), "saved_attempts": len(attempts),
        "failed_attempts": sum(a.get("status") != "success" for a in attempts),
        "attempts_with_usage": len(known), "attempts_without_usage": len(attempts)-len(known),
        "prompt_tokens": sum(u.get("prompt_tokens", 0) for u in known),
        "completion_tokens": sum(u.get("completion_tokens", 0) for u in known),
        "total_tokens": sum(u.get("total_tokens", 0) for u in known),
        "reasoning_tokens": sum((u.get("completion_tokens_details") or {}).get("reasoning_tokens", 0) for u in known),
        "request_seconds": round(sum(a.get("elapsed_seconds", 0) for a in attempts), 2),
    }


def stage(opid):
    if opid.endswith(".static.final"):
        return "static_generation"
    if ".doctor." in opid:
        return "interview_clinician"
    if ".patient." in opid:
        return "patient_sentence_order"
    if opid.endswith(".interactive.final"):
        return "interactive_final"
    if ".dialogue." in opid:
        return "dialogue_grading"
    return "final_answer_grading"


def main():
